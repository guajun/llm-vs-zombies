set_target_properties(recorder PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build")
